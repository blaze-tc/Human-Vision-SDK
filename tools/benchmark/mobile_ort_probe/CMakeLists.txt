cmake_minimum_required(VERSION 3.24)
project(hv_mobile_ort_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT ANDROID)
  message(FATAL_ERROR "Private Android device experiment only")
endif()
add_executable(hv_mobile_ort_probe main.cpp)
target_include_directories(hv_mobile_ort_probe PRIVATE "${ORT_ROOT}/include")
target_link_libraries(hv_mobile_ort_probe PRIVATE "${ORT_ROOT}/lib/libonnxruntime.so")
set_target_properties(hv_mobile_ort_probe PROPERTIES BUILD_RPATH "$ORIGIN")
find_package(ncnn REQUIRED)
add_executable(hv_mobile_ncnn_probe ncnn_probe.cpp)
target_link_libraries(hv_mobile_ncnn_probe PRIVATE ncnn)
